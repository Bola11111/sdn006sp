function doPost(e) {
  const result = { success: false, message: 'Invalid Action' };
  try {
    const params = JSON.parse(e.postData.contents);
    const action = params.action;
    const data = params.data;

    switch (action) {
      case 'login':
        result.data = loginHandle(data.username, data.password);
        result.success = !!result.data;
        if(!result.success) result.message = "Username atau Password salah.";
        break;
      case 'getAllData':
        result.data = fetchAllData();
        result.success = true;
        break;
      case 'getData':
        result.data = getSheetData(data.sheetName);
        result.success = true;
        break;
      case 'saveData':
        const saveRes = saveData(data.sheetName, data.formData, data.user);
        result.success = saveRes.success;
        result.message = saveRes.message;
        break;
      case 'deleteData':
        const delRes = deleteData(data.sheetName, data.id, data.user);
        result.success = delRes.success;
        result.message = delRes.message;
        break;
      case 'prosesMutasi':
        const mutRes = prosesMutasiKeluar(data.idSiswa, data.formData, data.user);
        result.success = mutRes.success;
        result.message = mutRes.message;
        break;
      case 'tarikDataMutasi':
        const tarikRes = tarikDataMutasi(data.idSiswa, data.user);
        result.success = tarikRes.success;
        result.message = tarikRes.message;
        break;
      case 'prosesKelulusan':
        const lulusRes = prosesKelulusanKelas6(data.tanggalLulus, data.user);
        result.success = lulusRes.success;
        result.message = lulusRes.message;
        result.count = lulusRes.count;
        break;
      case 'importData':
        result.success = importDataCSV(data.sheetName, data.csvData, data.user);
        result.message = result.success ? "Data berhasil diimpor" : "Gagal impor";
        break;
      case 'cetakSurat':
        const cetakRes = cetakSuratMutasi(data.id, data.jenis);
        result.success = cetakRes.success;
        result.message = cetakRes.message;
        result.url = cetakRes.url;
        break;
    }
  } catch (err) {
    result.message = err.toString();
  }

  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function loginHandle(username, password) {
  const data = getSheetData('Pengguna');
  const user = data.find(u => u.username === username && u.password === password);
  return user ? user : null;
}

function fetchAllData() {
  const ss = SpreadsheetApp.openById(SS_ID);
  const sheets = ['Tabel', 'Sekolah', 'Siswa', 'Pegawai', 'MutasiMasuk', 'MutasiKeluar', 'SiswaLulus', 'Pengguna'];
  const result = {};
  
  sheets.forEach(sheetName => {
    const sheet = ss.getSheetByName(sheetName);
    if(sheet) {
      const data = sheet.getDataRange().getDisplayValues();
      if(data.length > 1) {
        const headers = data.shift();
        result[sheetName] = data.map(row => {
          let obj = {};
          headers.forEach((h, i) => obj[h.toLowerCase().replace(/\s+/g, '_')] = row[i]);
          return obj;
        });
      } else {
        result[sheetName] = [];
      }
    } else {
      result[sheetName] = [];
    }
  });
  return result;
}

function getSheetData(sheetName) {
  const ss = SpreadsheetApp.openById(SS_ID);
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  const data = sheet.getDataRange().getDisplayValues(); 
  if (data.length <= 1) return [];
  
  const headers = data.shift();
  return data.map(row => {
    let obj = {};
    headers.forEach((h, i) => obj[h.toLowerCase().replace(/\s+/g, '_')] = row[i]);
    return obj;
  });
}

function saveData(sheetName, formData, user) {
  const ss = SpreadsheetApp.openById(SS_ID);
  const sheet = ss.getSheetByName(sheetName);
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const dataRows = sheet.getDataRange().getDisplayValues();
  
  let id = formData.id;
  let isUpdate = false;
  let rowIdx = -1;

  if (id) {
    const idIdx = headers.findIndex(h => h.toLowerCase() === 'id');
    for (let i = 1; i < dataRows.length; i++) {
      if (dataRows[i][idIdx].toString() === id.toString()) {
        rowIdx = i + 1;
        isUpdate = true;
        break;
      }
    }
  } else {
    id = Utilities.getUuid();
    formData.id = id;
  }

  let dataChanged = false; // DIRTY CHECKING FLAG

  if (sheetName === 'MutasiMasuk' || sheetName === 'MutasiKeluar' || sheetName === 'SiswaLulus') {
     formData['timestamp'] = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss");
  }

  const newRow = headers.map((h, colIdx) => {
    const key = h.toLowerCase().replace(/\s+/g, '_');
    let val = formData[key];
    
    if (isUpdate && val === undefined) {
      return dataRows[rowIdx - 1][colIdx]; 
    }

    if (val === undefined || val === null) val = "";
    val = val.toString().trim();
    
    // SMART FIELD: Proteksi Dinamis untuk semua angka berawalan Nol di SEMUA menu
    if (val.match(/^[0-9]+$/) && val.startsWith("0")) {
        val = "'" + val;
    } else if (val === "0" || val === "-") {
        val = "'" + val; 
    } else if (val.match(/^[0-9]+$/) && val.length > 5 && !val.startsWith("'")) {
        val = "'" + val; 
    }
    
    // Cek apakah data inputan berbeda dengan data yang ada di database
    if (isUpdate && val.toString() !== dataRows[rowIdx - 1][colIdx].toString()) {
        dataChanged = true;
    }

    return val;
  });

  if (isUpdate) {
    // EKSEKUSI HANYA JIKA ADA PERUBAHAN
    if (dataChanged) {
        sheet.getRange(rowIdx, 1, 1, headers.length).setValues([newRow]);
    } else {
        // Jika tidak ada perubahan, langsung return sukses tanpa eksekusi berat
        return { success: true, message: "Tidak ada perubahan data." };
    }
  } else {
    sheet.appendRow(newRow);
    rowIdx = sheet.getLastRow();
    dataChanged = true; // Data baru pasti dianggap changed
  }

  // AUTO SORT (Siswa & Pegawai) HANYA jika ada Data Baru/Perubahan
  if (dataChanged) {
    if (sheetName === 'Siswa' && sheet.getLastRow() > 1) {
      sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).sort([{column: 2, ascending: true}, {column: 3, ascending: true}]);
    } else if (sheetName === 'Pegawai' && sheet.getLastRow() > 1) {
      sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).sort([{column: 2, ascending: true}]);
    }
  }

  // Catat Log
  if (dataChanged) {
    let detailInfo = formData.nama || formData.nama_sekolah || formData.username || id;
    let actionName = isUpdate ? 'Memperbarui' : 'Menambah';
    catatLog(user, `${actionName} Data ${sheetName}`, `Data: ${detailInfo}`);
  }

  // ==========================================
  // AUTO GENERATE SURAT MUTASI SAAT SIMPAN
  // ==========================================
  if (sheetName === 'MutasiMasuk' || sheetName === 'MutasiKeluar') {
    try {
      const module = sheetName === 'MutasiMasuk' ? 'mutasi-masuk' : 'mutasi-keluar';
      const obj = {};
      headers.forEach((h, i) => obj[h.toString().toLowerCase().replace(/\s+/g, '_')] = newRow[i]);
      
      const docResult = cetakDokumenMutasi(module, obj);
      
      if (docResult.success) {
        const linkSuratColIdx = headers.findIndex(h => h.toString().toLowerCase().replace(/\s+/g, '_') === 'link_surat') + 1;
        if (linkSuratColIdx > 0) {
          sheet.getRange(rowIdx, linkSuratColIdx).setValue(docResult.url);
        }
      } else {
        console.error("Gagal Auto Generate Surat: " + docResult.message);
      }
    } catch(err) {
      console.error(err);
    }
  }

  return { success: true, message: "Data berhasil disimpan!" };
}

// HELPER: Ekstrak ID File Drive dari URL
function extractDriveFileId(url) {
  if (!url) return null;
  let match = url.match(/\/d\/(.+?)\//);
  if (match && match[1]) return match[1];
  match = url.match(/id=([^&]+)/);
  if (match && match[1]) return match[1];
  return null;
}

function deleteData(sheetName, id, user) {
  const ss = SpreadsheetApp.openById(SS_ID);
  const sheet = ss.getSheetByName(sheetName);
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const idIdx = headers.findIndex(h => h.toLowerCase() === 'id');
  
  for (let i = 1; i < data.length; i++) {
    if (data[i][idIdx].toString() === id.toString()) {
      
      const namaIdx = headers.findIndex(h => h.toLowerCase() === 'nama');
      let detail = namaIdx > -1 ? data[i][namaIdx] : id;

      // DRIVE CLEANER: Cek jika ada file dokumen yang tersangkut sebelum baris dihapus
      if (sheetName === 'MutasiMasuk' || sheetName === 'MutasiKeluar') {
        const linkIdx = headers.findIndex(h => h.toLowerCase().replace(/\s+/g, '_') === 'link_surat');
        if (linkIdx > -1 && data[i][linkIdx]) {
          const fileId = extractDriveFileId(data[i][linkIdx]);
          if (fileId) {
            try { DriveApp.getFileById(fileId).setTrashed(true); } catch(e) { console.warn("Gagal hapus file drive: " + e); }
          }
        }
      }

      sheet.deleteRow(i + 1);
      
      catatLog(user, `Menghapus Data ${sheetName}`, `Identitas Terhapus: ${detail}`);
      
      return { success: true, message: "Terhapus!" };
    }
  }
  return { success: false, message: "ID tidak ditemukan" };
}

function prosesMutasiKeluar(idSiswa, dataMutasi, user) {
  const siswa = getSheetData('Siswa').find(s => s.id === idSiswa);
  if (!siswa) return { success: false, message: "Siswa tidak ditemukan" };
  
  const combined = { ...siswa, ...dataMutasi };
  const res = saveData('MutasiKeluar', combined, user);
  
  if (res.success) {
    deleteData('Siswa', idSiswa, user);
    catatLog(user, `Proses Mutasi Keluar`, `Siswa: ${siswa.nama}`);
    return { success: true, message: "Berhasil memutasikan siswa & Dokumen disiapkan." };
  }
  return res;
}

function tarikDataMutasi(idSiswa, user) {
  const mutasi = getSheetData('MutasiKeluar').find(s => s.id === idSiswa);
  if (!mutasi) return { success: false, message: "Data mutasi tidak ditemukan" };
  
  const res = saveData('Siswa', mutasi, user);
  if (res.success) {
    // Fungsi deleteData akan otomatis menjalankan Drive Cleaner
    deleteData('MutasiKeluar', idSiswa, user);
    catatLog(user, `Tarik Batal Mutasi`, `Siswa: ${mutasi.nama}`);
    return { success: true, message: "Data berhasil ditarik kembali ke Siswa Aktif" };
  }
  return res;
}

// --- START PROSES KELULUSAN KELAS 6 (BATCH PROCESSING / MASSAL) --- //
function prosesKelulusanKelas6(tanggalLulus, user) {
  try {
    const ss = SpreadsheetApp.openById(SS_ID);
    const sheetSiswa = ss.getSheetByName('Siswa');
    const sheetLulus = ss.getSheetByName('SiswaLulus');
    
    if (!sheetSiswa || !sheetLulus) return { success: false, message: "Sheet Siswa atau SiswaLulus tidak ditemukan." };

    const dataSiswa = sheetSiswa.getDataRange().getDisplayValues();
    if (dataSiswa.length <= 1) return { success: false, message: "Tidak ada data siswa." };

    const headerSiswa = dataSiswa[0];
    const headerLulus = sheetLulus.getRange(1, 1, 1, sheetLulus.getLastColumn()).getValues()[0];
    
    const rombelIdx = headerSiswa.findIndex(h => h.toString().toLowerCase() === 'rombel');
    if (rombelIdx === -1) return { success: false, message: "Kolom Rombel tidak ditemukan." };

    // Map Index Header Siswa untuk pencarian instan (O(1) lookup)
    const headerSiswaIdxMap = {};
    headerSiswa.forEach((h, i) => headerSiswaIdxMap[h.toLowerCase().replace(/\s+/g, '_')] = i);

    const ts = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss");
    
    // Array penampung massal
    const barisDipertahankan = [];
    const barisKelulusan = [];
    
    // Looping data cukup di memori aplikasi (Sangat Cepat)
    for (let i = 1; i < dataSiswa.length; i++) {
      const row = dataSiswa[i];
      const rombel = row[rombelIdx].toString().trim().toUpperCase();
      
      if (rombel.includes('6') || rombel.includes('VI')) {
        // Susun baris baru mengikuti urutan header SiswaLulus secara presisi
        const newRow = headerLulus.map(h => {
          const key = h.toLowerCase().replace(/\s+/g, '_');
          if (key === 'tanggal_lulus') return tanggalLulus;
          if (key === 'timestamp') return ts;
          
          const sIdx = headerSiswaIdxMap[key];
          if (sIdx !== undefined) {
            let val = row[sIdx].toString().trim();
            if (val === "0" || val === "-") return "'" + val; 
            if (val.match(/^[0-9]+$/) && val.length > 5 && !val.startsWith("'")) val = "'" + val; 
            return val;
          }
          return "";
        });
        barisKelulusan.push(newRow);
      } else {
        barisDipertahankan.push(row);
      }
    }

    if (barisKelulusan.length > 0) {
      // 1. Eksekusi Massal: Sisipkan ke SiswaLulus dalam 1 API call
      sheetLulus.getRange(sheetLulus.getLastRow() + 1, 1, barisKelulusan.length, headerLulus.length).setValues(barisKelulusan);
      
      // 2. Eksekusi Massal: Bersihkan Siswa dan tulis ulang yang tersisa dalam 2 API call
      const lastRowSiswa = sheetSiswa.getLastRow();
      if (lastRowSiswa > 1) {
        sheetSiswa.getRange(2, 1, lastRowSiswa - 1, sheetSiswa.getLastColumn()).clearContent();
      }
      if (barisDipertahankan.length > 0) {
        sheetSiswa.getRange(2, 1, barisDipertahankan.length, headerSiswa.length).setValues(barisDipertahankan);
      }
      
      catatLog(user, `Proses Kelulusan Kelas 6`, `Sebanyak ${barisKelulusan.length} siswa diluluskan massal pada ${tanggalLulus}`);
      return { success: true, count: barisKelulusan.length, message: `${barisKelulusan.length} Siswa Kelas 6 berhasil diluluskan secara massal!` };
    } else {
      return { success: false, count: 0, message: "Tidak ada siswa Kelas 6 yang ditemukan." };
    }

  } catch (err) {
    return { success: false, message: err.toString() };
  }
}

function importDataCSV(sheetName, csvArray, user) {
  try {
    const ss = SpreadsheetApp.openById(SS_ID);
    const sheet = ss.getSheetByName(sheetName);
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => h.toLowerCase().replace(/\s+/g, '_'));
    
    const newRows = [];
    csvArray.forEach(rowObj => {
      let rowData = headers.map(key => {
        let val = rowObj[key] || rowObj[key.toUpperCase()] || rowObj[key.replace(/_/g, ' ')] || "";
        val = val.toString().trim();
        
        if (key === 'id' && val === "") return Utilities.getUuid();
        if (val.match(/^[0-9]+$/) && val.startsWith("0")) return "'" + val; // Smart Field Proteksi
        if (val === "0" || val === "-") return "'" + val; 
        if (val.match(/^[0-9]+$/) && val.length > 5 && !val.startsWith("'")) val = "'" + val;
        return val;
      });
      newRows.push(rowData);
    });

    if (newRows.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, newRows.length, headers.length).setValues(newRows);
      if (sheetName === 'Siswa') sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).sort([{column: 2, ascending: true}, {column: 3, ascending: true}]);
      if (sheetName === 'Pegawai') sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).sort([{column: 2, ascending: true}]);
      
      catatLog(user, `Import Excel ke ${sheetName}`, `Sebanyak ${newRows.length} baris data`);
    }
    return true;
  } catch(e) { return false; }
}

function cetakDokumenMutasi(module, rowData) {
  try {
    const templateId = module === 'mutasi-masuk' ? TEMPLATE_MASUK_ID : TEMPLATE_KELUAR_ID;
    
    if(!templateId || templateId.includes("MASUKKAN_ID")) return { success: false, message: "ID Template belum disetting di Code.gs" };
    if(!FOLDER_CETAK_ID || FOLDER_CETAK_ID.includes("MASUKKAN_ID")) return { success: false, message: "ID Folder belum disetting di Code.gs" };

    const folder = DriveApp.getFolderById(FOLDER_CETAK_ID);
    const templateDoc = DriveApp.getFileById(templateId);
    
    let namaSiswa = rowData['nama'] || rowData['nama_siswa'] || 'TanpaNama';
    if(typeof namaSiswa === 'string') namaSiswa = namaSiswa.replace(/^'/, '');
    
    let nisnSiswa = rowData['nisn'] || '000'; 
    if(typeof nisnSiswa === 'string') nisnSiswa = nisnSiswa.replace(/^'/, '');

    const jenisMutasi = module === 'mutasi-masuk' ? 'mutasi_masuk' : 'mutasi_keluar';
    const docName = `${namaSiswa.replace(/[^a-zA-Z0-9\s]/g, "")}_${nisnSiswa}_${jenisMutasi}`;
    
    const existingFiles = folder.getFilesByName(docName);
    while (existingFiles.hasNext()) { existingFiles.next().setTrashed(true); }
    
    const newFile = templateDoc.makeCopy(docName, folder);
    
    try {
      newFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    } catch (shareError) {
      console.warn("Izin setSharing dibatasi, namun file berhasil dibuat.", shareError);
    }

    const doc = DocumentApp.openById(newFile.getId());
    const body = doc.getBody();
    
    for (let key in rowData) {
      const placeholder = `<${key}>`;
      let nilaiKolom = rowData[key] || '';
      
      if (typeof nilaiKolom === 'string' && nilaiKolom.startsWith("'")) {
          nilaiKolom = nilaiKolom.substring(1);
      }
      
      if (key.includes('tanggal') || key.includes('tgl') || key.includes('tmt')) {
         nilaiKolom = formatTanggalIndo(nilaiKolom);
      }
      
      try { body.replaceText(placeholder, String(nilaiKolom)); } catch(e){}
    }
    
    doc.saveAndClose();
    return { success: true, url: newFile.getUrl() };
  } catch(e) {
    return { success: false, message: e.toString() };
  }
}

function cetakSuratMutasi(id, jenis) {
  const sheetName = jenis === 'masuk' ? 'MutasiMasuk' : 'MutasiKeluar';
  const ss = SpreadsheetApp.openById(SS_ID);
  const sheet = ss.getSheetByName(sheetName);
  const dataRows = sheet.getDataRange().getDisplayValues();
  const headers = dataRows[0];

  let rowIdx = -1;
  let rowData = null;
  const idIdx = headers.findIndex(h => h.toString().toLowerCase() === 'id');
  
  for(let i=1; i<dataRows.length; i++) {
    if(dataRows[i][idIdx].toString() === id.toString()) {
      rowIdx = i + 1;
      rowData = dataRows[i];
      break;
    }
  }
  if(!rowData) return { success: false, message: "Data mutasi tidak ditemukan pada sheet." };

  const obj = {};
  headers.forEach((h, i) => obj[h.toString().toLowerCase().replace(/\s+/g, '_')] = rowData[i]);

  const module = jenis === 'masuk' ? 'mutasi-masuk' : 'mutasi-keluar';
  const result = cetakDokumenMutasi(module, obj);

  if (result.success) {
     const linkSuratColIdx = headers.findIndex(h => h.toString().toLowerCase().replace(/\s+/g, '_') === 'link_surat') + 1;
     if(linkSuratColIdx > 0) sheet.getRange(rowIdx, linkSuratColIdx).setValue(result.url);
     return { success: true, message: "Surat berhasil dibuat manual.", url: result.url };
  } else {
     return { success: false, message: result.message };
  }
}

function formatTanggalIndo(dateStr) {
  if(!dateStr) return '';
  const parts = dateStr.split('-');
  if(parts.length === 3) {
    const bulanIndo = ['Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni', 'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember'];
    const d = parseInt(parts[2], 10);
    const m = parseInt(parts[1], 10) - 1;
    const y = parts[0];
    if (d && !isNaN(m) && y) return `${d} ${bulanIndo[m]} ${y}`;
  }
  return dateStr;
}
